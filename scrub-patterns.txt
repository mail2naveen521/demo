# =============================================================================
# Lines matching these regexes are removed before hashing files.
# Prevents harmless rewrites (timestamps, generated IDs) from being flagged
# as drift. One regex per line; # starts a comment.
# Tested against: WebSphere ND 8.5.5 + HCL Portal 9.5 on RHEL 8.
# =============================================================================

# --- WebSphere 8.5.5: regenerated identifiers --------------------------------
# xmi:id values are rewritten on most config saves and after server restarts
xmi:id="[^"]*"

# --- Timestamps in .properties files -----------------------------------------
# Default Java Properties file header: "#Sun Jan 01 12:00:00 UTC 2024"
^#[A-Z][a-z]{2} [A-Z][a-z]{2} [0-9 ][0-9] [0-9:]+ [A-Z]{3} [0-9]{4}$

# Variant without explicit timezone (some Portal saves)
^#[A-Z][a-z]{2} [A-Z][a-z]{2} [0-9 ][0-9] [0-9:]+ [0-9]{4}$

# --- Timestamps embedded in XML comments -------------------------------------
<!--.*[0-9]{4}-[0-9]{2}-[0-9]{2}T[0-9:]+.*-->
<!--.*Last [Mm]odified.*-->

# --- Generated UUIDs ---------------------------------------------------------
[a-f0-9]{8}-[a-f0-9]{4}-[a-f0-9]{4}-[a-f0-9]{4}-[a-f0-9]{12}

# --- WAS modification banner -------------------------------------------------
^#.*Modified on.*$
^#.*Last [Uu]pdated.*$

# --- Portal 9.5: ConfigWizard and ConfigEngine save markers ------------------
^# Generated by ConfigWizard.*$
^# Generated by HCL Portal ConfigEngine.*$

# Epoch milliseconds embedded in service property files
\.lastModified=[0-9]{10,}

# --- Portal 9.5: WCM and JCR runtime cache markers --------------------------
# These appear in some shared/app/config files after a node sync
cacheTimestamp=[0-9]+
syncTimestamp=[0-9]+

# --- WAS plugin-cfg.xml timestamps (regenerate on app deploy) ---------------
<!-- Plug-in Configuration generated on .* -->

# --- HCL DX xmlaccess export header -----------------------------------------
# Every xmlaccess export stamps the run time into a comment, e.g.:
#   <!--HCL Digital Experience/8.5 build 20250314-152926 exported on Sat May 16  14:34:00 EDT 2026 -->
# We strip ONLY the "exported on <timestamp>" clause, NOT the build string,
# so a genuine Portal binary/build-level change is still detected as drift.
exported on [A-Z][a-z]{2} [A-Z][a-z]{2} +[0-9]{1,2} +[0-9]{2}:[0-9]{2}:[0-9]{2} +[A-Z]{2,5} +[0-9]{4}

# Some CF levels emit an ISO-ish export stamp instead; cover that too
exported on [0-9]{4}-[0-9]{2}-[0-9]{2}[T ][0-9:]+([.+Z0-9-]*)?

# xmlaccess also writes an export-time attribute on the root <request>/<portal>
# element in some CFs: create-date / export-time style attributes
(create-date|export-time|exportTime|timestamp)="[^"]*"
